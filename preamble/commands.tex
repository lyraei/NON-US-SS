% ---------- Rozdział (wstęp, posłowie) ----------
% \rozdzial{Wstęp}
\newcommand{\rozdzial}[1]{%
    \cleardoublepage\thispagestyle{plain}%
    \markright{#1}\phantomsection
    \addcontentsline{toc}{section}{#1}%
    \bs@naglowek{#1}}

\newcommand{\engtytul}[1]{%
    \cleardoublepage\thispagestyle{plain}%
    \markright{#1}\phantomsection
    \addcontentsline{toc}{section}{#1}%
    \bs@naglowek{#1}}    

% ---------- Tier ----------
% \tier{I}{Filary}
\newcommand{\tier}[2]{%
    \cleardoublepage\thispagestyle{plain}%
    \markright{#2}\phantomsection
    \addcontentsline{toc}{section}{\texorpdfstring{#1\hspace{0.6em}#2}{#1. #2}}%
    \vspace*{12mm}
    {\numerfont #1\par}
    \vspace{-4mm}
    \bs@naglowek{#2}}

% Ścieżka względem img/; brak pliku → szary placeholder
\newcommand{\okladka}[1]{%
    \IfFileExists{img/#1}%
    {\includegraphics[width=\okladkaszer,height=\okladkaszer,keepaspectratio]{#1}}%
    {{\setlength{\fboxsep}{0pt}%
            \colorbox{szary1}{\parbox[c][\okladkaszer][c]{\okladkaszer}%
                {\centering\scriptsize\color{szary2}okładka}}}}}

\newenvironment{album}[4]{%
    \par\addvspace{\albumodstep}%

    \hspace{\okladkaodstep}%
    \begin{minipage}[t]{\dimexpr\linewidth-\okladkaszer-\okladkaodstep\relax}\vspace{-0.3ex}%
        \RaggedRight
        {\artystafont #2\par}
        {\tytulfont #3\par}
        \vspace{0.4ex}{\metafont #4\par}
        \vspace{1.4ex}\small
}{%
    \end{minipage}\par}
