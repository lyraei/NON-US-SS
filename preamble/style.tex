% Wygląd dokumentu: kolory, kroje, pagina, strona tytułowa, spis treści.
\makeatletter

% ---------- Akapit ----------
\linespread{1.18}
\setlength{\parindent}{1.1em}
\setlength{\parskip}{0pt}

% ---------- Kolory (monochrom) ----------
\definecolor{szary1}{gray}{0.93} % tła, placeholdery
\definecolor{szary2}{gray}{0.55} % metadane, transliteracje
\definecolor{szary3}{gray}{0.35} % pagina, linie

% ---------- Style tekstu ----------
\newcommand{\etykieta}{\addfontfeatures{LetterSpace=12, RawFeature=+case}\MakeUppercase}
\newcommand{\numerfont}{\displayfont\fontsize{40}{40}\selectfont\color{szary2}}
\newcommand{\tierfont}{\displayfont\bfseries\fontsize{22}{26}\selectfont}
\newcommand{\artystafont}{\bfseries\small}
\newcommand{\tytulfont}{\itshape}
\newcommand{\metafont}{\footnotesize\color{szary2}}

% ---------- Dane dokumentu ----------
\newcommand{\tytul}[1]{\def\bs@tytul{#1}}
\newcommand{\podtytul}[1]{\def\bs@podtytul{#1}}
\newcommand{\autor}[1]{\def\bs@autor{#1}}
\newcommand{\rok}[1]{\def\bs@rok{#1}}
\tytul{}\podtytul{}\autor{}\rok{}

% ---------- Pagina ----------
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyhead[LE]{\footnotesize\color{szary3}\etykieta{\bs@tytul}}
\fancyhead[RO]{\footnotesize\color{szary3}\etykieta{\rightmark}}
\fancyfoot[LE,RO]{\footnotesize\color{szary3}\thepage}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[LE,RO]{\footnotesize\color{szary3}\thepage}}

% ---------- Strona tytułowa ----------
\newcommand{\stronatytulowa}{%
    \begin{titlepage}
        \RaggedRight
        \vspace*{30mm}
        {\displayfont\bfseries\fontsize{28}{32}\selectfont \bs@tytul\par}
        \vspace{5mm}
        {\color{szary3}\rule{24mm}{0.5pt}\par}
        \vspace{5mm}
        {\large\color{szary3}\bs@podtytul\par}
        \vfill
        {\small\etykieta{\bs@autor}\par}
        \vspace{1mm}
        {\small\color{szary2}\bs@rok\par}
    \end{titlepage}
    \cleardoublepage}

% ---------- Spis treści ----------
\addto\captionspolish{\renewcommand{\contentsname}{Spis płyt}}
\renewcommand{\cfttoctitlefont}{\tierfont}
\renewcommand{\cftaftertoctitle}{\par\vspace{2mm}{\color{szary3}\rule{20mm}{0.4pt}}}
\setlength{\cftbeforetoctitleskip}{18mm}
\setlength{\cftaftertoctitleskip}{6mm}
\renewcommand{\cftsecfont}{\footnotesize\bfseries\addfontfeatures{LetterSpace=6}}
\renewcommand{\cftsecpagefont}{\footnotesize\bfseries}
\renewcommand{\cftsecleader}{\hfill}
\setlength{\cftbeforesecskip}{3.5mm}
\renewcommand{\cftsubsecfont}{\footnotesize}
\renewcommand{\cftsubsecpagefont}{\footnotesize\color{szary2}}
\renewcommand{\cftsubsecleader}{\hfill}
\setlength{\cftsubsecindent}{0pt}
\setlength{\cftbeforesubsecskip}{0.6mm}

% ---------- Nagłówek rozdziału ----------
\newcommand{\bs@naglowek}[1]{%
    \vspace*{18mm}
    {\RaggedRight\tierfont #1\par}
    \vspace{2mm}{\color{szary3}\rule{20mm}{0.4pt}\par}
    \vspace{8mm}\@afterindentfalse\@afterheading}

\makeatother
